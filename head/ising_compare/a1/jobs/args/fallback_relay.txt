# A1's BB72 fallback answers (revision/REVISION_PLAN.md 2.1, "Pair curves"): full-DEM Relay-BP under A6's policy
# (section 2.6: a new decoder per shot, seeded with the shot's row in the set; a1.fallback relay) on every a1test shot,
# once. A6 has not run, so A1 runs the policy itself (revision/A1_READ_PLAN.md). relay-bp 0.2.2 lives in bb's venv
# (the phase C runs' $PYR), not in the torch venv, so this job calls that venv's python in the same container. About
# 70 ms a shot: 500,000 shots are about 10 core-hours, about 10-15 minutes on 64 cores.
# DRY=1: the dry test set (a1dry, chunks 4-9) into $A/reads_dry_main/bb72.
#   sbatch -p acpu --qos=cpu-normal -c 64 --mem=64G -t 01:00:00 -J a1_relay --export=ALL,CMD=fallback_relay.txt a1/jobs/job.sbatch
source a1/jobs/args/common.sh
set -x
PYR="apptainer exec -B <cluster-scratch> <cluster-scratch>/ising/pyfull.sif <cluster-scratch>/bb/venv/bin/python"
if [ -n "${DRY:-}" ]; then K="--kind a1dry --chunks 4-9"; R=$A/reads_dry_main/bb72; else K="--kind a1test --chunks 0-49"; R=$A/reads/bb72; fi
mkdir -p $R
$PYR -m a1.fallback relay --code bb72 $K --out $R/fallback_test.npz --workers ${SLURM_CPUS_PER_TASK:-32}
