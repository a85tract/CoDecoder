# CPU (one whole amem node with an AMD EPYC 7713, exclusive, about 1 h; after the freeze, with E1's routing masks at the
# frozen tau pulled to $E/pod/masks/frozen): the CPU term of every network, new and deployed, as the studies ran it: three
# interleaved repetitions, each a fresh single-threaded process on core 8, of the node's correlated PM per block on
# validation chunk 0 and of every network's term on its routed validation shots -> ${CPU:-$E/cpu} (analysis: the
# slowest term times G(d) over the median ref). Run from a deployment of F or B: the networks are surf_v2.session's, so
# the freeze record's exceptions decide them (a SKIP, if set, must name the same networks). The job refuses a directory
# that exists: a failed job runs again into a fresh one (CPU=$E/cpu2, and so on), the first job that completes counts
# (surf_v2/REGISTRATION.md, Timing), and alpine_analysis.txt takes the same CPU.
#   sbatch -p amem --qos=mem-normal --nice=1000 -N 1 --exclusive --mem=256G --constraint=epyc-7713 -t 03:00:00 -J sv2e_cpu ...
[ "${SLURM_JOB_PARTITION:-}" = amem ] && scontrol show job $SLURM_JOB_ID | grep -q "OverSubscribe=NO" &&
  grep -q "AMD EPYC 7713 64-Core" /proc/cpuinfo || { echo "not one exclusive amem node with an EPYC 7713: stop"; exit 6; }
echo "core 8 siblings $(cat /sys/devices/system/cpu/cpu8/topology/thread_siblings_list); load $(cat /proc/loadavg)"
C=${CPU:-$E/cpu}; M=${MASKS:-$E/pod/masks/frozen}
[ -e surf_v2/freeze.json ] || { echo "no surf_v2/freeze.json: the CPU term runs from a deployment of F or B"; exit 7; }
[ ! -e $C ] || { echo "$C exists: a CPU-term job writes into a fresh directory (CPU=...)"; exit 7; }
NETS=$($PY -c "from surf_v2 import session; print(' '.join(f'{d}:{n}' for d, n in session.networks()))"); mkdir -p $C
for r in 1 2 3; do
  for d in 7 9 11 13 15 17; do $PY -m timing.cpu_term ref --d $d --val $E/shots/val --rep $r --core 8 --out $C; done
  for x in $NETS; do d=${x%%:*}; n=${x#*:}
    $PY -m timing.cpu_term term --d $d --val $E/shots/val --mask $M/val_$n.npz --rep $r --core 8 --out $C; done
done
(cd $C && sha256sum *.json)
